\section{Tether range}

\begin{frame}{The tether is held taut by the gravity gradient}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\small
Circular orbit of radius $a = R_\oplus + h$:
\begin{equation*}
n = \sqrt{\frac{\mu}{a^{3}}} = 1.0948\times10^{-3}\ \mathrm{rad/s},\qquad P = \frac{2\pi}{n} = 95.6\ \mathrm{min}.
\end{equation*}
A body forced to turn at rate $n$ at radius $a+l$ lacks
\begin{equation*}
n^{2}(a+l) - \frac{\mu}{(a+l)^{2}} \simeq 3\,n^{2}\,l
\end{equation*}
of inward acceleration. The tether supplies it:
\begin{equation*}
T = 3\,n^{2}\,\meq\,L,\qquad \meq = \frac{m_1 m_2}{m_1+m_2} = 2\ \mathrm{kg}.
\end{equation*}
\vspace{0.1cm}
\textbf{$T = 3.6$\,mN (500\,m) \quad $T = 7.2$\,mN (1\,km)}
\end{column}
\begin{column}{0.4\textwidth}
\centering

\begin{frame}{Only the vertical direction separates the two halves}
\small
Relative motion in the orbiting frame ($x$ radial, $y$ along track, $z$ out of plane):
\begin{equation*}
\ddot{x} - 2n\dot{y} - 3n^{2}x = f_x,\qquad
\ddot{y} + 2n\dot{x} = f_y,\qquad
\ddot{z} + n^{2}z = f_z .
\end{equation*}
\vspace{-0.2cm}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}lll@{}}
\toprule
Direction & Free motion & Tether \\
\midrule
Radial ($x$) & $+3n^{2}x$: the halves move apart & taut \\
Along track ($y$) & no restoring or separating term & slack \\
Out of plane ($z$) & $-n^{2}z$: the halves move together & slack \\
\bottomrule
\end{tabular}
\end{center}
A tilt $\varphi$ from the vertical swings back (constant $L$):
\begin{equation*}
\ddot{\varphi} + 3n^{2}\sin\varphi\cos\varphi = 0
\;\Rightarrow\;
\omega_{\mathrm{lib}} = \sqrt{3}\,n,\quad 2\pi/\omega_{\mathrm{lib}} = 55.2\ \mathrm{min}.
\end{equation*}
\alert{A diagonal tether is a swing, not a configuration.} Holding $\varphi = 30^\circ$ at 1\,km needs
$3n^{2}\meq L^{2}\sin\varphi\cos\varphi = 3.1$\,N\,m, about 3000 times a CubeSat reaction wheel.
\end{frame}

% =====================================================================
\begin{frame}{500 m to 1 km with one tether: the key numbers}
\centering
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lccc@{}}
\toprule
Quantity & Expression & $L = 500$\,m & $L = 1$\,km \\
\midrule
Tension & $3n^{2}\meq L$ & 3.6\,mN & 7.2\,mN \\
Elastic stretch & $TL/(EA)$ & 0.6\,mm & 2.3\,mm \\
Thermal length change, $\Delta\theta = 100$\,K & $|\alpha|\,L\,\Delta\theta$ & 0.15 to 0.5\,m & 0.3 to 1.0\,m \\
Tether mass, $\varnothing$ 0.2\,mm & $\rho_c A L$ & 22\,g & 44\,g \\
Tether frontal area & $d\,L$ & 0.10\,m$^{2}$ & 0.20\,m$^{2}$ \\
Controlled deployment, $\lambda = n/2$ & $\ln(L/L_0)/\lambda$ & 3.2\,h & 3.5\,h \\
Controlled retrieval, $\lambda = n/5$ & $\ln(L/L_0)/\lambda$ & 7.9\,h & 8.8\,h \\
Orbit life at 550\,km, deployed & drag model & $\approx 2.5$\,yr & $\approx 1.6$\,yr \\
\bottomrule
\end{tabular}

\vspace{0.25cm}
{\footnotesize\color{black!65} $EA = 3142$\,N, $\alpha \approx -10^{-5}$/K (polyethylene) to $-3\times10^{-6}$/K (aramid), $L_0 = 1$\,m, mean solar activity. Stowed: $\approx 6$\,yr.}
\end{frame}

\begin{frame}{Deployment has to be slow}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
In-plane dynamics with tether length $L$ and tilt $\varphi$:
\begin{align*}
\ddot{L} &= L\big[(\dot{\varphi}+n)^{2} - n^{2}(1-3\cos^{2}\varphi)\big] - \frac{T}{\meq},\\
\ddot{\varphi} &= -2\,\frac{\dot{L}}{L}\,(\dot{\varphi}+n) - 3n^{2}\sin\varphi\cos\varphi .
\end{align*}
The term $-2\dot{L}/L$ (Coriolis) tilts the tether while it pays out.
\end{column}
\begin{column}{0.45\textwidth}
\textbf{Free, no brake} ($T=0$, $\varphi\equiv0$):
\begin{equation*}
L(t) = L_0\cosh\!\big(\sqrt{3}\,n\,t\big)
\end{equation*}
75 to 81\,min, but ends at 0.95 to 1.9\,m/s and the tether tips over.

\vspace{0.2cm}
\textbf{Controlled}, $L = L_0\,e^{\lambda t}$:
\begin{equation*}
\sin 2\varphi^{*} = -\frac{4\lambda}{3n},\qquad \lambda \le \tfrac{3}{4}\,n .
\end{equation*}
With $\lambda = n/2$: $\varphi^{*} = -20.9^\circ$ and 3.2 to 3.5\,h.
\end{column}
\end{columns}
\end{frame}

% =====================================================================
\begin{frame}{Once deployed, the length stays fixed}
The winch sets $L$. More tension does not lengthen the tether; it only pulls it tighter.
\vspace{0.3cm}
\begin{columns}[T]
\begin{column}{0.31\textwidth}
\begin{block}{Elastic stretch}
\centering
$\Delta L_{\mathrm{el}} = \dfrac{T\,L}{EA}$\\[4pt]
{\Large\bfseries $\le$ 2.3\,mm}\\[2pt]
\footnotesize at 1\,km and full tension
\end{block}
\end{column}
\begin{column}{0.31\textwidth}
\begin{block}{Thermal change}
\centering
$\Delta L_{\mathrm{th}} = |\alpha|\,L\,\Delta\theta$\\[4pt]
{\Large\bfseries up to 1\,m}\\[2pt]
\footnotesize sun to shadow; the winch compensates
\end{block}
\end{column}
\begin{column}{0.31\textwidth}
\begin{block}{Natural swing}
\centering
$2\pi/(\sqrt{3}\,n)$\\[4pt]
{\Large\bfseries 55\,min}\\[2pt]
\footnotesize period around the vertical
\end{block}
\end{column}
\end{columns}
\vspace{0.4cm}
The tension is fixed by the length: $T = 3n^{2}\meq L$. Paying out faster only adds swing and braking load.
\end{frame}

\section{Relative attitude}

\begin{frame}{Relative attitude: IMUs alone are not enough}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\small
\textbf{Why the stowed zero is lost}
\begin{itemize}
\item Gyros integrate the rate, so a bias $b$ grows:
$\delta\theta(t) \simeq b\,t$.
\item $b = 1^\circ$/h gives $1.6^\circ$ after one orbit.
\item Accelerometers cannot sense attitude in free fall.
\item One tether does not stop the halves from turning about its own axis.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\small
\textbf{What to fly on each half}
\begin{itemize}
\item \textbf{Star tracker}: absolute attitude, $\approx 0.005^\circ$.
\begin{equation*}
R_{AB} = R_A^{\top}R_B
\end{equation*}
\item \textbf{GNSS}: tether direction from the baseline
\begin{equation*}
\hat{u} = \frac{\mathbf{r}_B - \mathbf{r}_A}{\lVert\mathbf{r}_B - \mathbf{r}_A\rVert},\quad \delta\hat{u} \approx \frac{\delta r}{L} \approx 10^{-5}\ \mathrm{rad}
\end{equation*}
\item \textbf{Tether encoder} at each anchor; \textbf{IMU} for fast rates.
\end{itemize}
\end{column}
\end{columns}
\vspace{0.2cm}
\vspace{0.05cm}
{\small\textbf{Stowed zero:} calibrates the mounting offset between the halves only.}
\end{frame}

\begin{column}{0.42\textwidth}
A point of height $h$ shifts between the photos by
\begin{equation*}
p = h\,\frac{B}{H},
\end{equation*}
so the height error is
\begin{equation*}
\sigma_h = \frac{\sigma_p}{B/H}.
\end{equation*}
$B$ baseline, $H = 550$\,km, $\sigma_p$ matching error
(0.3 pixel $=$ 1.5\,m with 5\,m pixels).
\end{column}

\begin{column}{0.42\textwidth}
Drag decay of the orbit radius:
\begin{equation*}
\dot{a} = -\rho\,C_D\,\frac{A}{m}\,\sqrt{\mu a},\qquad A = A_{\mathrm{sat}} + d\,L .
\end{equation*}
A 1\,km tether of 0.2\,mm adds 0.20\,m$^{2}$, three times the satellites.
\begin{itemize}
\item \textbf{Reel in between campaigns}: about 6\,yr instead of 1.6\,yr.
\item \textbf{Deploy at end of life} to reenter faster: at 590\,km, 3\,yr instead of 11.5\,yr.
\end{itemize}
\end{column}

\begin{frame}{Other risks}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{@{}>{\bfseries}p{3.0cm}p{10.4cm}@{}}
\toprule
Retrieval & Reeling in amplifies the swing ($-2\dot{L}/L > 0$). It needs libration control and a slow rate: 7.9 to 8.8\,h. \\
Thermal length & Up to 1\,m per orbit at 1\,km. The winch needs that range and closed-loop control. \\
Single tether & One cut by a small debris particle ends the mission. Use a braided or multi-strand line. \\
Debris and admissibility & A 1\,km object has a larger collision cross-section. Confirm with the CSA. \\
Model accuracy & Orbit lives come from a simple drag model; a formal STELA or DRAMA study is needed. \\
\bottomrule
\end{tabular}
\end{frame}
